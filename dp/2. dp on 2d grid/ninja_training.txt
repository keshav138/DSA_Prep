Plan----
1. Given a 2d array of n days and 3 tasks everyday, one can do one task everyday and each has some rewards to offer, and one task done yestarday
cannot be done today.
2. Task is to maximise rewards by choosing which task to do everyday.

Soln----------
1. Our parameters are the day we start from, we start from the last day to the 0th day, we carry a last day task parameter, to track what was the
last track done, and dp array for memoization.
2. For our base case, ie on the last day, we simple choose the max value out of the three tasks as our ans.
3. Then for the function on each call, we loop over the 0-2 for the tasks, if done previously we skip, else we consider the tasks reward,
and recusively call for the next day, updating the last done task.


-----------------CODE---------------------------
class Solution {
  public:
    
    int solve(int day, int last_task, vector<vector<int>>&mat, vector<vector<int>>&dp){
        if(day == 0){
            int cur_max = -1;
            for(int task=0; task<=2; task++){
                if(task==last_task) continue;
                cur_max = max(cur_max, mat[day][task]);
            }
            return cur_max;
        }
        
        if(dp[day][last_task]!=-1) return dp[day][last_task];
        
        int temp_max = -1;
        for(int task=0; task<=2; task++){
            if(task==last_task) continue;
            temp_max = max(temp_max, mat[day][task] + solve(day-1, task, mat, dp));
        }
        
        return dp[day][last_task] = temp_max; 
    }
    
    int maximumPoints(vector<vector<int>>& mat) {
        // code here
        int n = mat.size();
        vector<vector<int>> dp(n, vector<int>(4, -1));
        return solve(n-1, 3, mat, dp);
    }
};
