1. Pretty simple, take the path with the min sum, just a few conditions here and there
2. Since calculating for min, sometimes the call may go out of bound where earlier we were returning 0, this turns out to be a problem because
we're taking the min of both calls, we need a indicator saying that 'hey this call did not make it to the source', for that instead of returning a
0(indicating a null path), we return 1e9, a huge value so that the min fuction avoids that specific call.

class Solution {
public:

    int solve(int m, int n, vector<vector<int>>&dp, vector<vector<int>>&grid){
        if(m==0 && n==0) return grid[m][n];
        if(m<0 || n<0) return 1e9; // the special return

        if(dp[m][n]!=-1) return dp[m][n];

        int up = grid[m][n] + solve(m-1, n, dp, grid);
        int left = grid[m][n] + solve(m, n-1, dp, grid);

        return dp[m][n] = min(left, up);
    }

    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();

        vector<vector<int>> dp(m, vector<int>(n,-1));

        return solve(m-1, n-1, dp, grid);
    }
};
