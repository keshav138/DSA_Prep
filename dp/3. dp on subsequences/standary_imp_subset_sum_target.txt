bool f(int idx, int target, vector<int>&arr, vector<vector<int>>&dp){
    if(target == 0) return true;
    if(idx == 0) return (arr[idx] == target);
    if(dp[idx][target]!=-1) return dp[idx][target];

    bool notPick = f(idx-1, target, arr, dp);
    bool Pick = false;

    if(arr[idx] >= target) Pick = f(idx-1, target-arr[idx], arr, dp);

    return dp[idx][target] = Pick | notPick; ## boolean converts to 0/1 automatically
}



bool subsetSum(int target, vector<int>&arr){
    n = arr.size()
    vector<vector<int>> dp(n, vector<int>(target+1, -1));

    return f(n-1, target, arr, dp);
}
