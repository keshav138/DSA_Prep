Concept is similar to calculating longest common subsequence, only difference is we only have one case, where we find the palindrome, where we do a +1, and compare a max.
due to the constraint of substrings.

class Solution {
  public:
    int longCommSubstr(string& s1, string& s2) {
        // your code here
        int n = s1.length();
        int m = s2.length();
        
        vector<vector<int>>dp(n+1, vector<int>(m+1, 0));
        int res = -1e9;
        
        for(int i=1; i<=n; i++){
            for(int j=1; j<=m; j++){
                if(s1[i-1] == s2[j-1]) dp[i][j] = 1 + dp[i-1][j-1];
                res = max(res, dp[i][j]);
            }
        }
        
        return res;
    }
};
