Tree BFS Traversal----------

vector<vector<int>> bfs(Node* root){
    if(root == nullptr) return {};

    queue<Node *> q;
    vector<vector<int>> res;

    q.push(root);
    int currLevel = 0;


    while(!q.emtpy()){
        int len = q.size();
        res.push_back({});

        for(int i=0; i<len; i++){
            Node* fr = q.front();
            q.pop();

            res[currLevel].push_back(fr->data);

            if(fr->left!=nullptr) q.push_back(fr->left); 
            if(fr->right!=nullptr) q.push_back(fr->right); 

        }
        currLevel++;
 
    }
    return res;
}
