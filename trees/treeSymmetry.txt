/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:

    bool ishelp(TreeNode* leftRoot, TreeNode* rightRoot){
        ## simple if both null return trueo

        if(leftRoot == nullptr && rightRoot == nullptr) return true;

        ## if one of them is null, or the values dont match, mind the order, first check for null's then only compare values

        if(leftRoot == nullptr || rightRoot == nullptr || leftRoot->val != rightRoot->val) return false;
        
        ## check for remaining tree
        return ishelp(leftRoot->left, rightRoot->right) && ishelp(leftRoot->right, rightRoot->left);
    }

    bool isSymmetric(TreeNode* root) {
        if(root == nullptr) return true;

        return ishelp(root->left, root->right);
    }
};
