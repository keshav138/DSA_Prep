Plan----
1. Simple, given two grids, if a island in grid 2 is a subset of island blocks in grid 1, its a sub island.
2. So we iterate on the grid 2, if we find a land block we run a dfs run.
3. We set a flag, on doing a vis = true, we check if the current block was a land block in grid 1, if its a water block, grid 2 island cant be a sub island, set the flag to false, if not continue.
4. If on returning we have a true flag, we increment the counter.

----


class Solution {
public:

    void dfs(int row, int col, vector<vector<int>>&grid1, vector<vector<int>>&grid2, vector<vector<bool>>&vis, bool &flag){
        vis[row][col] = true;
        if(grid1[row][col] == 0) flag = false;

        int r = grid1.size();
        int c = grid1[0].size();

        int rdir[] = {0, -1, 0, 1};
        int cdir[] = {-1, 0, 1, 0};

        for(int i=0; i<4; i++){
            int nrow = row + rdir[i];
            int ncol = col + cdir[i];

            if(nrow > -1 && nrow < r && ncol > -1 && ncol < c && !vis[nrow][ncol] && grid2[nrow][ncol]==1){
                dfs(nrow, ncol, grid1, grid2, vis, flag);
            }
        }
    }



    int countSubIslands(vector<vector<int>>& grid1, vector<vector<int>>& grid2) {
        int rows = grid1.size();
        int cols = grid1[0].size();

        vector<vector<bool>>vis(rows, vector<bool>(cols, false));


        int counter = 0;
        for(int i=0; i<rows; i++){
            for(int j=0; j<cols; j++){
                if(grid2[i][j] == 1 && !vis[i][j]){
                    bool flag = true;
                    dfs(i, j, grid1, grid2, vis, flag);
                    if (flag) counter++;
                }
            }
        }

        return counter;
    }
};
