Plan------
1. Given a 2d rxc matrix, with O's and X's, we're to convert all 0's which are surrounded with X's on all directions,
left, top, right, bottom. O's include a group of 0's or individual O's.

2. All the boundary O's are invalid since they are not surrounded, so any O's connected to these boundary O's are also
invalid, therefore we run a dfs through all the boundary O's with a visited matrix(rxc size).

3. All O's that are untouched are O's that can be safely surrounded by X's therefore converted to X'safely


---------CODE------------
class Solution {
public:

    void dfs(int row, int col, vector<vector<char>>&board, vector<vector<int>>&vis, int r, int c){
        vis[row][col] = 1;

        int dr[] = {0, -1, 0, 1};
        int dc[] = {-1, 0, 1, 0};

        for(int i=0; i<4; i++){
            int nrow = row + dr[i];
            int ncol = col + dc[i];

            if(nrow >= 0 && nrow < r && ncol >= 0 && ncol < c && !vis[nrow][ncol] && board[nrow][ncol] == 'O'){
                dfs(nrow, ncol, board, vis, r, c);
            }
        }
    }

    void solve(vector<vector<char>>& board) {
        int r = board.size();
        int c = board[0].size();

        vector<vector<int>> vis(r, vector<int>(c, 0));
        
        for(int i = 0; i<c; i++){
            if(board[0][i] == 'O' && !vis[0][i]) dfs(0, i, board, vis, r, c);
            if(board[r-1][i] == 'O' && !vis[r-1][i]) dfs(r-1, i, board, vis, r, c);
        }

        for(int i=0; i<r; i++){
            if(board[i][0] == 'O' && !vis[i][0]) dfs(i, 0, board, vis, r, c);
            if(board[i][c-1] == 'O' && !vis[i][c-1]) dfs(i, c-1, board, vis, r, c);
        }

        for(int i=0; i<r; i++){
            for(int j=0; j<c; j++){
                if(vis[i][j] == 0 && board[i][j]=='O'){
                    board[i][j] = 'X';
                }
            }
        }
    }
};
