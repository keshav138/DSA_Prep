1. Basically we have a 2d grid, where we want to find the number of land boxes(marked by one), which have no connection to the boundaries
2. In simple words, we use teh similar approach as earlier in some questions, we visit all land nodes on the edges and run a bfs through them.
3. In the end whichever land box is not visited we take those counts:



-----CODE---------
class Solution {
public:
    int numEnclaves(vector<vector<int>>& grid) {
        queue<pair<int, int>>q;

        int r = grid.size();
        int c = grid[0].size();
        vector<vector<int>> vis(r, vector<int>(c,0));
        // int vis[r][c] = {0};

        for(int i=0; i<r; i++){
            for(int j=0; j<c; j++){
                // ## first row , last row, last col, first col
                if(i==0 || i==r-1 || j==0 || j==c-1){
                    if(grid[i][j]==1){
                        q.push({i,j});
                        vis[i][j] = 1;
                    }
                }
            }
        }

        while(!q.empty()){
            int row = q.front().first;
            int col = q.front().second;
            q.pop();

            int drow[] = {0, -1, 0, 1};
            int dcol[] = {-1, 0, 1, 0};

            for(int i=0; i<4; i++){
                int curr_row = row + drow[i];
                int curr_col = col + dcol[i];

                if(curr_row>=0 && curr_row<r && curr_col>=0 && curr_col<c && grid[curr_row][curr_col]==1 && vis[curr_row][curr_col]==0){
                    q.push({curr_row, curr_col});
                    vis[curr_row][curr_col] = 1;
                }
            }
        }

        int count = 0;
        for(int i=0; i<r; i++){
            for(int j=0; j<c; j++){
                if(grid[i][j] == 1 && vis[i][j]==0) count++;
            }
        }

        return count;
    }
};
