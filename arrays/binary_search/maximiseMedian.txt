
bool canAchieveMedian(int mid, int n, int k , vector<int>&arr){
    long long cost = 0;
    int median_idx = n/2;


    for(int i = median_idx; i<n; i++){
        if(mid > arr[i]){
            cost += mid - arr[i];
            if(cost > k) return false;
        }
    }

    return cost <= k;
}





long long solve(){
    int n;
    long long k;

    cin>>n>>k;

    vector<int> arr(n);

    for(int i=0; i<n; i++){
        cin>>arr[i];
    }

    sort(arr.begin(), arr.end());

    long long low = arr[n/2];
    long long high = arr[n/2] + k;

    long long ans = low;

    while(low <= high){
        long long mid = low + (high - low) / 2;

        if(canAchieveMedian(mid, n, k, arr)){
            ans = mid;
            low = mid+1;
        }
        else{
            high = mid-1;
        }
    }

    return ans;
}
