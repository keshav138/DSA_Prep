class Solution {
public:
    int minOperations(vector<int>& nums, int x) {
        int sum = 0; 
        for(auto i : nums) sum += i;


        int target = sum - x;
        if(sum == 0) return nums.size();
        if(sum < 0) return -1;

        int curr_sum = 0;
        int ans = -1;
        int left = 0;
        for(int right=0; right<nums.size(); right++){

            curr_sum += nums[right];

            while(curr_sum > target && left <= right){
                curr_sum -= nums[left];
                left++;
            }
            
            if(curr_sum == target){
                ans = max(ans, right - left + 1);
            }

        }

        if(ans  == -1) return -1;
        else return nums.size() - ans;


    }
};
