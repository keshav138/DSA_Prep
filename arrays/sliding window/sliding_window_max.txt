The deque property is the only one that matters here, we pop from front those values that dont belong in the window,
and pop from back the values when the monotonic property is faulted by an incoming value, we pop until we can insert our
incoming element.

As soon as we hit tht min window length, we start push values from the front of the deque into the ans vector.

some index so look out for: 
i-k is validating window size, example i=6, window size = 3, 6-3 = 3, so our valid range is [4,5,6] so any value, below
and equal to i-k does not belong to the window.


class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int> dq;
        vector<int>ans;
        int n = nums.size();

        for(int i=0; i<n; i++){
            //check if the .first() are valid window elements else pop
            while(!dq.empty() && dq.front() <= i-k){
                dq.pop_front();
            }

            // checking if we can push if property fails we pop back
            while(!dq.empty() && nums[dq.back()] < nums[i]){
                dq.pop_back();
            }

            dq.push_back(i);

            // if we've entered valid window size, now every iteration gives us an ans
            if(i >= k-1){
                ans.push_back(nums[dq.front()]);
            }
        }
        return ans;
    }
};
